function getItemName(code) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Item_Master");
  const data = sheet.getRange(2, 1, sheet.getLastRow() - 1, 3).getValues();
  for (let i = 0; i < data.length; i++) {
    if ((data[i][0] + '').trim() === (code + '').trim()) {
      return data[i][2];
    }
  }
  return null;
}

function generateBOM(fgCode, plannedQty) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const bomSheet = ss.getSheetByName("BOM");
  const itemMasterSheet = ss.getSheetByName("Item_Master");
  const draftSheet = ss.getSheetByName("Issue_Draft");

  const fgCodes = bomSheet.getRange("H16:KJ16").getValues()[0].map(code => (code + '').trim());
  const itemCodes = bomSheet.getRange("A17:A1060").getValues().flat().map(code => (code + '').trim());
  const bomMatrix = bomSheet.getRange("H17:KJ1060").getValues();
  const itemMaster = itemMasterSheet.getRange(2, 1, itemMasterSheet.getLastRow() - 1, 3).getValues();

  const fgIndex = fgCodes.findIndex(code => code === fgCode.trim());
  if (fgIndex === -1) {
    throw new Error(`FG Code "${fgCode}" not found in BOM.`);
  }

  draftSheet.clearContents();
  draftSheet.appendRow(["Item Code", "Item Description", "Qty per pc", "Total RC Qty", "Issued Qty"]);

  for (let i = 0; i < itemCodes.length; i++) {
    const qtyPerPc = parseFloat(bomMatrix[i][fgIndex]);
    if (!qtyPerPc || isNaN(qtyPerPc)) continue;

    const itemCode = itemCodes[i];
    const itemRow = itemMaster.find(row => (row[0] + '').trim() === itemCode);
    const itemDesc = itemRow ? itemRow[2] : "";

    draftSheet.appendRow([itemCode, itemDesc, qtyPerPc, qtyPerPc * plannedQty, ""]);
  }
}

function confirmIssue(date, routeCard) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const issueDraft = ss.getSheetByName('Issue_Draft');
  const storesOut = ss.getSheetByName('Stores_Out');

  const draftData = issueDraft.getRange(2, 1, issueDraft.getLastRow() - 1, 5).getValues();
  const rawHeader = storesOut.getRange(1, 1, 1, storesOut.getLastColumn()).getValues()[0];

  const [yyyy, mm, dd] = date.split("-");
  const inputDate = new Date(`${yyyy}-${mm}-${dd}`);
  inputDate.setHours(0, 0, 0, 0);

  let dateColIndex = -1;
  for (let i = 0; i < rawHeader.length; i++) {
    const cellValue = rawHeader[i];
    if (Object.prototype.toString.call(cellValue) === "[object Date]") {
      const headerDate = new Date(cellValue);
      headerDate.setHours(0, 0, 0, 0);
      if (headerDate.getTime() === inputDate.getTime()) {
        dateColIndex = i + 1;
        break;
      }
    }
  }

  if (dateColIndex === -1) {
    SpreadsheetApp.getUi().alert(`Date ${dd}/${mm}/${yyyy} not found in Stores_Out sheet header.`);
    return;
  }

  const itemCodesInStores = storesOut.getRange(2, 1, storesOut.getLastRow() - 1, 1).getValues().flat();
  for (let i = 0; i < draftData.length; i++) {
    const [itemCode, , , , issuedQty] = draftData[i];
    if (!itemCode || !issuedQty || isNaN(issuedQty)) continue;

    const rowIndex = itemCodesInStores.indexOf(itemCode) + 2;
    if (rowIndex < 2) continue;

    const cell = storesOut.getRange(rowIndex, dateColIndex);
    const existingValue = Number(cell.getValue()) || 0;
    const existingComment = cell.getComment() || '';

    const newQty = existingValue + Number(issuedQty);
    const newComment = existingComment.includes(routeCard)
      ? existingComment
      : existingComment
      ? `${existingComment}, ${routeCard}`
      : routeCard;

    cell.setValue(newQty);
    cell.setComment(newComment);
  }
}

function submitFGConsumption(data) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const bomSheet = ss.getSheetByName("BOM");
  const shortageSheet = ss.getSheetByName("Shortage_april");

  const fgCode = (data.fgCode || "").trim();
  const producedQty = parseFloat(data.fgQty);
  const routeCard = (data.routeCard || "").trim();
  const date = data.date;

  Logger.log("fgCode: " + fgCode);
  Logger.log("producedQty: " + producedQty);
  Logger.log("routeCard: " + routeCard);
  Logger.log("date: " + date);

  if (!fgCode || isNaN(producedQty) || producedQty <= 0) {
    SpreadsheetApp.getUi().alert("Invalid FG Code or Produced Qty.");
    return;
  }

  const fgCodes = bomSheet.getRange("H16:KJ16").getValues()[0];
  const itemCodes = bomSheet.getRange("A17:A1060").getValues().flat();
  const bomMatrix = bomSheet.getRange("H17:KJ1060").getValues();

  const fgIndex = fgCodes.findIndex(code => (code + '').trim() === fgCode);
  if (fgIndex === -1) {
    SpreadsheetApp.getUi().alert(`FG Code "${fgCode}" not found in BOM.`);
    return;
  }

  const shortageItemCodes = shortageSheet.getRange(2, 1, shortageSheet.getLastRow() - 1).getValues().flat();

  for (let i = 0; i < itemCodes.length; i++) {
    const itemCode = (itemCodes[i] + '').trim();
    const bomQtyPerPc = parseFloat(bomMatrix[i][fgIndex]);
    if (!itemCode || isNaN(bomQtyPerPc)) continue;

    const totalConsumed = bomQtyPerPc * producedQty;

    const rowIndex = shortageItemCodes.findIndex(code => (code + '').trim() === itemCode);
    if (rowIndex === -1) continue;

    const cell = shortageSheet.getRange(rowIndex + 2, 12); // L column
    const currentValue = parseFloat(cell.getValue());
    const updatedValue = (isNaN(currentValue) ? 0 : currentValue) + totalConsumed;

    cell.setValue(updatedValue);
  }

  SpreadsheetApp.getUi().alert(`FG Consumption for ${fgCode} logged successfully in Column AP.`);
}
